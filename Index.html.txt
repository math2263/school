    function getKSTDateString(dateObj = new Date()) {
      const utc = dateObj.getTime() + (dateObj.getTimezoneOffset() * 60000);
      const kstTime = new Date(utc + (9 * 3600000));
      const year = kstTime.getFullYear();
      const month = String(kstTime.getMonth() + 1).padStart(2, '0');
      const day = String(kstTime.getDate()).padStart(2, '0');
      return `${year}-${month}-${day}`;
    }

    // 대한민국 표준시(KST) 기준 오늘 연도(YYYY) 반환
    function getKSTYear(dateObj = new Date()) {
      const utc = dateObj.getTime() + (dateObj.getTimezoneOffset() * 60000);
      const kstTime = new Date(utc + (9 * 3600000));
      return kstTime.getFullYear();
    }

    // 대한민국 표준시(KST) 기준 오늘 월(YYYY-MM) 반환
    function getKSTMonthString(dateObj = new Date()) {
      const utc = dateObj.getTime() + (dateObj.getTimezoneOffset() * 60000);
      const kstTime = new Date(utc + (9 * 3600000));
      const year = kstTime.getFullYear();
      const month = String(kstTime.getMonth() + 1).padStart(2, '0');
      return `${year}-${month}`;
    }

    function loadYearData(year) {
      if (year === currentOperatingYear) {
        isReadOnlyArchive = false;
        const savedTeachers = localStorage.getItem('sr_teachers');
        const savedStudents = localStorage.getItem('sr_students');
        const savedRecords = localStorage.getItem('sr_records');

        if (savedTeachers) {
          teachers = JSON.parse(savedTeachers);
          teachers.forEach(t => { if (t.id === 'admin') t.id = '관리자'; });
        } else {
          teachers = [
            { id: '관리자', pw: '1234', name: '관리자', role: 'admin', assignedClass: '' },
            { id: 'teacher1', pw: '1234', name: '김담임', role: 'teacher', assignedClass: '105' },
            { id: 'sub_teacher', pw: '1234', name: '이교사', role: 'teacher', assignedClass: '' }
          ];
          saveTeachers();
        }

        if (!teachers.some(t => t.role === 'admin')) {
          teachers.unshift({ id: '관리자', pw: '1234', name: '관리자', role: 'admin', assignedClass: '' });
          saveTeachers();
        }

        students = savedStudents ? JSON.parse(savedStudents) : [];
        records = savedRecords ? JSON.parse(savedRecords) : [];
      } else {
        isReadOnlyArchive = true;
        const archData = archives[year] || { students: [], teachers: [], records: [] };
        students = archData.students || [];
        teachers = archData.teachers || [{ id: '관리자', pw: '1234', name: '관리자', role: 'admin', assignedClass: '' }];

    function renderClassGroupOptions() {
      const classSelect = document.getElementById('classGroupSelect');
      const current = classSelect.value;
      const groups = [...new Set(students.map(s => s.id.substring(0, s.id.length - 2)))].sort();

      classSelect.innerHTML = '<option value="">학년·반 선택...</option>';
      groups.forEach(g => {
        const opt = document.createElement('option');
        opt.value = g;
        opt.textContent = `${g} (${g.substring(0,1)}학년 ${parseInt(g.substring(1))}반)`;
        classSelect.appendChild(opt);
      });
      if (current && groups.includes(current)) classSelect.value = current;
    }

    function onClassGroupChange() {
      const g = document.getElementById('classGroupSelect').value;
      const numSelect = document.getElementById('numberSelect');
      numSelect.innerHTML = '<option value="">번호 선택...</option>';

      if (!g) { numSelect.disabled = true; return; }
      const filtered = students.filter(s => s.id.startsWith(g)).sort((a,b) => a.id.localeCompare(b.id));
      filtered.forEach(s => {
        const opt = document.createElement('option');
        opt.value = s.id;
        opt.textContent = `${s.id.substring(s.id.length - 2)}번 (${s.name})`;
        numSelect.appendChild(opt);
      });
      numSelect.disabled = false;
      renderStudentHistory();
    }

    function onSearchInput() {
      const q = document.getElementById('studentSearchInput').value.trim().toLowerCase();
      if (!q) return;
      const match = students.find(s => s.id.toLowerCase().includes(q) || s.name.toLowerCase().includes(q));
      if (match) selectStudent(match.id, false);
    }

    function selectStudent(id, updateInput = true) {
      selectedStudentId = id;
      const student = students.find(s => s.id === id);
      if (!student) return;

      const groupVal = student.id.substring(0, student.id.length - 2);
      const classSelect = document.getElementById('classGroupSelect');
      if (classSelect.value !== groupVal) {
        classSelect.value = groupVal;
        onClassGroupChange();
      }
      document.getElementById('numberSelect').value = student.id;
      if (updateInput) document.getElementById('studentSearchInput').value = student.id;

      updateSelectedStudentUI();
      renderStudentHistory();
    }

    function updateSelectedStudentUI() {
      const banner = document.getElementById('selectedStudentBanner');
      if (!selectedStudentId) { banner.classList.add('hidden'); return; }
      const s = students.find(st => st.id === selectedStudentId);
      if (s) {
        banner.classList.remove('hidden');
        document.getElementById('bannerStudentId').textContent = s.id;
        document.getElementById('bannerStudentName').textContent = s.name;
        const pending = records.filter(r => r.studentId === s.id && !r.reflectionSubmitted).length;
        const badge = document.getElementById('bannerPendingBadge');
        if (pending > 0) { badge.textContent = `미제출 ${pending}건`; badge.classList.remove('hidden'); }
        else badge.classList.add('hidden');
      }
    }

    function confirmAndRecordViolation() {
      if (isReadOnlyArchive) { alert('과거 보관 데이터 조회 중에는 새로 입력할 수 없습니다.'); return; }
      if (!selectedStudentId) { alert('학생을 먼저 선택해 주세요.'); return; }
      if (selectedViolations.size === 0) { alert('위반 항목을 1개 이상 선택해 주세요.'); return; }

      const date = document.getElementById('violationDate').value;
      const noteInput = document.getElementById('violationNote');
      const note = noteInput.value.trim();
      const direct = document.getElementById('reflectionSubmittedDirectly').checked;

      if (selectedViolations.has('기타') && !note) { 
        alert("'기타' 항목 선택 시 세부 사유를 입력하세요."); 
        noteInput.focus(); 
        return; 
      }

      const nowString = new Date().toLocaleString('ko-KR', {
        timeZone: 'Asia/Seoul',
        year: 'numeric', month: '2-digit', day: '2-digit',
        hour: '2-digit', minute: '2-digit'
      });

      const violationTypes = Array.from(selectedViolations);
      violationTypes.forEach((vType, index) => {
        records.unshift({
          id: 'rec_' + Date.now() + '_' + index,
          studentId: selectedStudentId,
          date: date,
          type: vType,
          note: note,
          reflectionSubmitted: direct,
          teacherName: currentUser ? currentUser.name : '알 수 없음',
          createdAt: nowString
        });
      });

      selectedViolations.clear();
      renderViolationButtons();
      noteInput.value = '';
      document.getElementById('reflectionSubmittedDirectly').checked = false;
      setTodayToViolationDate();

      alert(`선택한 위반 항목 ${violationTypes.length}건이 각각 독립된 개별 건으로 정상 등록되었습니다.`);
      saveData();
    }

    function toggleReflection(id) {
      if (isReadOnlyArchive) { alert('과거 보관 데이터는 수정할 수 없습니다.'); return; }
      const rec = records.find(r => r.id === id);
      if (!rec) return;

      const studentClass = rec.studentId.substring(0, rec.studentId.length - 2);

      if (currentUser.role !== 'admin') {
        if (!currentUser.assignedClass) {
          alert('비담임 교사는 성찰문 제출 여부를 체크할 수 없습니다.');
          return;
        }
        if (currentUser.assignedClass !== studentClass) {
          alert(`권한이 없습니다. (${studentClass}반 담임교사만 체크 가능)`);
          return;
        }
      }

    function renderStats() {
      const monthVal = document.getElementById('statsMonthPicker').value;
      const filtered = records.filter(r => r.date.startsWith(monthVal));

      const cats = violationCategories;
      const maxCat = Math.max(...cats.map(c => filtered.filter(r => {
        if (Array.isArray(r.type)) return r.type.includes(c);
        return r.type === c;
      }).length), 1);

      document.getElementById('catChartContainer').innerHTML = cats.map(c => {
        const cnt = filtered.filter(r => {
          if (Array.isArray(r.type)) return r.type.includes(c);
          return r.type === c;
        }).length;
        const pct = (cnt / maxCat) * 100;
        return `
          <div class="bar-row">
            <span class="bar-label" title="${c}">${c}</span>
            <div class="bar-track"><div class="bar-fill" style="width:${pct}%;"></div></div>
            <span style="width:30px;">${cnt}건</span>
          </div>`;
      }).join('');

      const classMap = {};
      filtered.forEach(r => {
        const g = r.studentId.substring(0, r.studentId.length - 2);
        classMap[g] = (classMap[g] || 0) + 1;
      });
      const classKeys = Object.keys(classMap).sort();
      const maxClass = Math.max(...Object.values(classMap), 1);
      document.getElementById('classChartContainer').innerHTML = classKeys.length ? classKeys.map(k => {
        const cnt = classMap[k];
        const pct = (cnt / maxClass) * 100;
        return `
          <div class="bar-row">
            <span class="bar-label">${k}반</span>
            <div class="bar-track"><div class="bar-fill" style="width:${pct}%; background:linear-gradient(90deg, #6366f1 0%, #818cf8 100%);"></div></div>
            <span style="width:30px;">${cnt}건</span>
          </div>`;
      }).join('') : '<div style="text-align:center; color:#94a3b8; padding:20px;">기록 없음</div>';

      const active = students.map(s => {
        const sRecs = filtered.filter(r => r.studentId === s.id);
        const sub = sRecs.filter(r => r.reflectionSubmitted).length;
        const ratio = sRecs.length ? ((sub / sRecs.length) * 100).toFixed(1) : '0.0';

        const counts = {};
        sRecs.forEach(r => {
          const types = Array.isArray(r.type) ? r.type : [r.type];
          types.forEach(t => { if (t) counts[t] = (counts[t] || 0) + 1; });
        });
        let topCat = '-'; let max = 0;
        for (const [k,v] of Object.entries(counts)) { if (v > max) { max = v; topCat = k; } }

        return { id: s.id, name: s.name, total: sRecs.length, sub, ratio, topCat };
      }).filter(s => s.total > 0).sort((a,b) => b.total - a.total);

      document.getElementById('filteredStudentCountBadge').textContent = `${active.length}명`;
      const tbody = document.getElementById('statsStudentTableBody');
      if (!active.length) {
        tbody.innerHTML = '<tr><td colspan="6" style="text-align:center; padding:30px; color:#94a3b8;">해당 월에 위반 기록이 있는 학생이 없습니다.</td></tr>';
        return;
      }
      tbody.innerHTML = active.map(s => `
        <tr>
          <td><strong>${s.id}</strong></td>
          <td>${s.name}</td>
          <td style="color:#4338ca; font-weight:bold;">${s.total}회</td>
          <td style="color:#059669; font-weight:bold;">${s.sub}건 제출</td>
          <td>${s.ratio}% (${s.sub}/${s.total})</td>
          <td><span style="background:#f1f5f9; padding:2px 6px; border-radius:4px;">${s.topCat}</span></td>
        </tr>
      `).join('');
    }

      const id = document.getElementById('newTeacherId').value.trim();

      const name = document.getElementById('newTeacherName').value.trim();
      const role = document.getElementById('newTeacherRole').value;
      const assignedClass = document.getElementById('newTeacherClass').value.trim();

      if (teachers.some(t => t.id === id)) {
        alert('이미 존재하는 교사 아이디입니다.');
        return;
      }

      teachers.push({ id, pw, name, role, assignedClass });
      saveTeachers();
      renderTeacherList();

      document.getElementById('newTeacherId').value = '';
      document.getElementById('newTeacherPw').value = '';

    function processBatchTeacherRegister() {
      if (isReadOnlyArchive) { alert('과거 보관 데이터 조회 중에는 등록할 수 없습니다.'); return; }
      const text = document.getElementById('batchTeacherInputText').value.trim();
      if (!text) { alert('등록할 내용을 입력하세요.'); return; }

      let count = 0;
      text.split('\n').forEach(line => {
        const parts = line.split(/[,;\t]/).map(p => p.trim());
        if (parts.length >= 3 && parts[0] && parts[1] && parts[2] && parts[0] !== '아이디') {
          if (!teachers.some(t => t.id === parts[0])) {
            teachers.push({
              id: parts[0],
              pw: parts[1],
              name: parts[2],
              role: parts[3] || 'teacher',
              assignedClass: parts[4] || ''
            });
            count++;
          }
        }
      });
      saveTeachers();
      renderTeacherList();
      document.getElementById('batchTeacherInputText').value = '';
      alert(`총 ${count}명의 신규 교사 계정이 누적 등록되었습니다.`);
    }

    function deleteTeacher(id) {
      if (isReadOnlyArchive) { alert('과거 보관 데이터는 삭제할 수 없습니다.'); return; }
      if (id === '관리자') { alert('최고 관리자 계정은 삭제할 수 없습니다.'); return; }
      if (confirm(`⚠️ 경고: 교사 계정(${id})을 삭제하시겠습니까?\n\n교사 계정을 삭제할 경우 해당 교사가 기록한 지도 내역 등은 유지되나 계정 접근이 불가해집니다.`)) {
        teachers = teachers.filter(t => t.id !== id);
        selectedTeacherIds.delete(id);
        saveTeachers();
        renderTeacherList();
      }
    }

    function toggleTeacherSelection(id, checked) {
      if (checked) selectedTeacherIds.add(id);
      else selectedTeacherIds.delete(id);
      updateSelectedTeacherCountUI();
    }

    function toggleSelectAllTeachers(checked) {
      teachers.forEach(t => {
        if (t.id !== '관리자') {
          if (checked) selectedTeacherIds.add(t.id);
          else selectedTeacherIds.delete(t.id);
        }
      });
      document.querySelectorAll('.teacher-item-checkbox').forEach(cb => cb.checked = checked);
      updateSelectedTeacherCountUI();
    }

    function updateSelectedTeacherCountUI() {
      document.getElementById('selectedTeacherCount').textContent = selectedTeacherIds.size;
      const nonAdminTeachers = teachers.filter(t => t.id !== '관리자');
      const allCb = document.getElementById('selectAllTeachersCheckbox');
      if (allCb && nonAdminTeachers.length > 0) {
        allCb.checked = nonAdminTeachers.every(t => selectedTeacherIds.has(t.id));
      }
    }

    function deleteSelectedTeachers() {
      if (isReadOnlyArchive) { alert('과거 보관 데이터는 삭제할 수 없습니다.'); return; }
      if (selectedTeacherIds.size === 0) {
        alert('삭제할 교사를 체크박스로 선택해 주세요.');
        return;
      }

      if (confirm(`⚠️ 경고: 선택한 ${selectedTeacherIds.size}명의 교사 계정을 정말로 삭제하시겠습니까?`)) {
        teachers = teachers.filter(t => t.id === '관리자' || !selectedTeacherIds.has(t.id));
        selectedTeacherIds.clear();
        saveTeachers();
        renderTeacherList();
        alert('선택한 교사 계정이 일괄 삭제되었습니다.');
      }
    }

    function renderTeacherList() {
      const tbody = document.getElementById('teacherTableBody');
      tbody.innerHTML = teachers.map(t => {
        const isChecked = selectedTeacherIds.has(t.id);
        return `
          <tr>
            <td>${t.id !== '관리자' ? `<input type="checkbox" class="teacher-item-checkbox" style="accent-color:#4338ca;" ${isChecked ? 'checked' : ''} onchange="toggleTeacherSelection('${t.id}', this.checked)">` : ''}</td>
            <td><strong>${t.id}</strong></td>
            <td>${t.name}</td>
            <td><span class="user-badge ${t.role === 'admin' ? 'badge-admin' : (t.assignedClass ? 'badge-teacher' : 'badge-sub')}">${t.role === 'admin' ? '관리자' : (t.assignedClass ? '담임' : '비담임')}</span></td>
            <td>${t.assignedClass ? t.assignedClass + '반' : '비담임'}</td>
            <td>
              ${(t.id !== '관리자' && !isReadOnlyArchive) ? `<button class="btn btn-rose" style="padding:2px 8px; font-size:11px;" onclick="deleteTeacher('${t.id}')">삭제</button>` : '-'}
            </td>
          </tr>
        `;
      }).join('');
      updateSelectedTeacherCountUI();
    }

    // 엑셀 다중 시트 생성 및 다운로드 핵심 함수
    function exportMultiSheetExcel(filteredRecs, titlePrefix, filename) {
      if (!filteredRecs.length) { alert('출력할 데이터가 없습니다.'); return; }

      // 1시트: 개별 상세 지도 이력
      const sheet1Data = [
        ["학번", "이름", "날짜", "위반항목", "성찰문제출여부", "등록교사", "등록시각", "세부메모"]
      ];
      filteredRecs.forEach(r => {
        const s = students.find(st => st.id === r.studentId) || { name: '미등록' };
        const typeStr = Array.isArray(r.type) ? r.type.join(', ') : r.type;
        sheet1Data.push([
          r.studentId,
          s.name,
          r.date,
          typeStr,
          r.reflectionSubmitted ? '제출완료' : '미제출',
          r.teacherName || '',
          r.createdAt || '',
          r.note || ''
        ]);
      });

      // 2시트: 학생별 위반 집계
      const sheet2Data = [
        ["학번", "이름", "총 위반 횟수", "성찰문 제출 횟수", "성찰문 제출율(%)", "주요 위반 항목"]
      ];
      const studentMap = {};
      filteredRecs.forEach(r => {
        if (!studentMap[r.studentId]) {
          const s = students.find(st => st.id === r.studentId) || { name: '미등록' };
          studentMap[r.studentId] = { id: r.studentId, name: s.name, total: 0, sub: 0, cats: {} };
        }
        studentMap[r.studentId].total++;
        if (r.reflectionSubmitted) studentMap[r.studentId].sub++;
        const types = Array.isArray(r.type) ? r.type : [r.type];
        types.forEach(t => {
          if (t) studentMap[r.studentId].cats[t] = (studentMap[r.studentId].cats[t] || 0) + 1;
        });
      });

      const studentStatsArr = Object.values(studentMap).sort((a,b) => b.total - a.total);
      studentStatsArr.forEach(st => {
        const ratio = st.total ? ((st.sub / st.total) * 100).toFixed(1) : '0.0';
        let topCat = '-'; let maxCnt = 0;
        for (const [k,v] of Object.entries(st.cats)) {
          if (v > maxCnt) { maxCnt = v; topCat = k; }
        }
        sheet2Data.push([st.id, st.name, st.total, st.sub, `${ratio}%`, topCat]);
      });

      // 3시트: 학급별 위반 집계
      const sheet3Data = [
        ["학년·반", "총 위반 건수"]
      ];
      const classMap = {};
      filteredRecs.forEach(r => {
        const g = r.studentId.substring(0, r.studentId.length - 2);
        classMap[g] = (classMap[g] || 0) + 1;
      });
      const sortedClasses = Object.keys(classMap).sort();
      sortedClasses.forEach(cls => {
        sheet3Data.push([`${cls}반`, classMap[cls]]);
      });

      const wb = XLSX.utils.book_new();
      
      const ws1 = XLSX.utils.aoa_to_sheet(sheet1Data);
      const ws2 = XLSX.utils.aoa_to_sheet(sheet2Data);
      const ws3 = XLSX.utils.aoa_to_sheet(sheet3Data);

      XLSX.utils.book_append_sheet(wb, ws1, "상세 지도 이력");
      XLSX.utils.book_append_sheet(wb, ws2, "학생별 위반 집계");
      XLSX.utils.book_append_sheet(wb, ws3, "학급별 위반 집계");

      XLSX.writeFile(wb, filename);
    }
